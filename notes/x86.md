# SYS V

- stack:
  - stack grows down
  - 4-byte aligned on older systems; 16-byte aligned just before the call instruction is called
    - i think this means we can just be 16-byte aligned at all times, but doesn't that waste 75% of the space?
- functions:
  - function params are pushed in reverse order (param 3, param 2, param 1)
    - params can be modified by callee
  - callee should immediately push ebp onto the stack and then set ebp to that address
  - call pushes address of next instruction onto the stack and jumps to fn addr
  - ret pops address from stack and jumps to it
  - direction flag must be called before calling and returning from a function
- registers:
  - non-volatile:
    - ebx
    - esi
    - edi
    - ebp
  - volatile:
    - eax
    - ecx
    - edx
  - return register is eax; if the return value is 64 bits, the higher 32 bits go in edx

Example:

```asm
section .data
  fn_result: dd 0

section .text
main:
  push 123            ; push a couple things to the stack
  push 'a'

  ; stack before calling fn:
  ;
  ; ebp -> | 123                | <- main's stack data
  ; esp -> | 'a'                |

  call fn             ; call our function

  ; stack after calling fn:
  ;
  ; ebp -> | 123                | <- main's stack data
  ; esp -> | 'a'                |

  mov fn_result, eax  ; move the result to a very important global var

fn:
  ; stack at fn entry:
  ;
  ; ebp -> | 123                | <- main's stack data
  ;        | 'a'                |
  ; esp -> | fn's return addr   |

  push ebp     ; save caller's ebp
  mov ebp esp  ; init our ebp

  push 42      ; push something onto our stack

  ; stack:
  ;
  ;        | 123                | <- main's stack data
  ;        | 'a'                |
  ;        | fn's return addr   |
  ; ebp -> | main's ebp         | <- fn's stack data
  ; esp -> | 42                 |

  ; do some stuff

  mov eax, 1337 ; move 1337 to the return register

  mov esp ebp   ; clear fn stack

  ; stack after clearing fn stack:
  ;
  ;               | 123                | <- main's stack data
  ;               | 'a'                |
  ;               | fn's return addr   |
  ; esp -> ebp -> | main's ebp         | <- fn's stack data

  pop ebp ; restore ebp

  ; stack after restoring ebp:
  ;
  ; ebp -> | 123                | <- main's stack data
  ;        | 'a'                |
  ; esp -> | fn's return addr   |

  ret ; pop the next stack value and jump to it
```
